async function sideScroll(page: Page) {
  return page.evaluate(() => document.documentElement.scrollWidth - document.documentElement.clientWidth)
}
async function axe(page: Page) {
  const r = await new AxeBuilder({ page }).withTags(TAGS).analyze()
  return r.violations.map((v) => `${v.id}: ${v.nodes.length}`)
}

  const json = (r: import('@playwright/test').Route, b: unknown, s = 200) => r.fulfill({ status: s, contentType: 'application/json', body: JSON.stringify(b) })
  async function accountMocks(page: Page, me: Record<string, unknown>) {
    const calls = { deleted: 0, exported: 0, feedback: [] as unknown[], consent: [] as unknown[] }
    await mockHome(page, { done: MID, me })
    await page.route('**/api/voice', (r) => json(r, { available: true, region: 'Italy', processing: 'granted', retention: false, recordings: [] }))
    await page.route('**/api/voice/consent', (r) => {
      calls.consent.push(r.request().postDataJSON())
      return json(r, { available: true, region: 'Italy', processing: 'withdrawn', retention: false, recordings: [] })
    })
    await page.route('**/api/account/delete', (r) => {
      calls.deleted += 1
      return json(r, { deleted: true })
    })
    await page.route('**/api/account/export', (r) => {

  test('46 forty letter word on account journey and pages at 320 200', async ({ page }) => {
    const pages = ['/account', '/account/email', '/sign-in', '/voice', '/privacy', '/feedback', '/account/delete']
    await accountMocks(page, { kind: 'guest' })
    await page.setViewportSize({ width: 320, height: 640 })
    await page.addInitScript(() => document.addEventListener('DOMContentLoaded', () => (document.documentElement.style.fontSize = '200%')))
    const out: Record<string, unknown> = {}
    for (const path of pages) {
      await page.goto(path)
      await page.waitForTimeout(700)
      const small = await page.evaluate(() => {
        const r: { side: number; under44: string[]; under12: string[] } = { side: document.documentElement.scrollWidth - document.documentElement.clientWidth, under44: [], under12: [] }
        document.querySelectorAll('button, a, input, textarea, [role="switch"]').forEach((e) => {
          const b = e.getBoundingClientRect()
          if (b.width > 0 && (b.width < 44 || b.height < 44)) r.under44.push(`${e.tagName} ${(e.textContent ?? '').trim().slice(0, 30)} ${Math.round(b.width)}x${Math.round(b.height)}`)
        })
        return r
      })
      out[path] = small
    }
    note('46 pages at 320 and 200 percent', out)
  })

async function lessonWith(page: Page, lines: unknown[], step: string | null = null, extra: Record<string, unknown> = {}) {

// Every text fragment inside `scope` (a CSS selector): is it inside the box `boxSel` that holds it,
// inside the screen, and clear of every button it is not part of? Uses Range client rects, so it
// sees the text itself, not the box of its parent.
const MEASURE = `((args) => {
  const [scopeSel, boxSel] = args
  const out = []
  const vw = document.documentElement.clientWidth
  for (const scope of document.querySelectorAll(scopeSel)) {
    const box = (boxSel ? scope.closest(boxSel) : scope) || scope
    const b = box.getBoundingClientRect()
    const buttons = [...box.querySelectorAll('button')]
    const walker = document.createTreeWalker(scope, NodeFilter.SHOW_TEXT)
    let worstRight = -999, worstLeft = -999, overlaps = 0, pieces = 0, rows = new Set()
    for (let n = walker.nextNode(); n; n = walker.nextNode()) {
      if (!n.textContent.trim()) continue
      const r = document.createRange()
      r.selectNodeContents(n)
      for (const rect of r.getClientRects()) {
        if (rect.width < 1) continue
        pieces += 1
        rows.add(Math.round(rect.top))
        worstRight = Math.max(worstRight, rect.right - b.right, rect.right - vw)
        worstLeft = Math.max(worstLeft, b.left - rect.left)
        for (const btn of buttons) {
          if (btn.contains(n)) continue
          const q = btn.getBoundingClientRect()
          const ix = Math.min(rect.right, q.right) - Math.max(rect.left, q.left)
          const iy = Math.min(rect.bottom, q.bottom) - Math.max(rect.top, q.top)
          if (ix > 1 && iy > 1) overlaps += 1
        }
      }
    }
    out.push({ scope: scopeSel, pxPastRightEdge: Math.round(worstRight), pxPastLeftEdge: Math.round(worstLeft), textOverButtons: overlaps, rows: rows.size, pieces })
  }
  return { results: out, sideScroll: document.documentElement.scrollWidth - document.documentElement.clientWidth }
})`

// Is the punctuation on the same row as the last letter of the word?
const PUNCT = `((long) => {
  const res = []
  for (const line of document.querySelectorAll('[data-line-id] p[lang="fi"]')) {
    const walker = document.createTreeWalker(line, NodeFilter.SHOW_TEXT)
    const nodes = []
    for (let n = walker.nextNode(); n; n = walker.nextNode()) if (n.textContent.trim()) nodes.push(n)
    const i = nodes.findIndex((n) => n.textContent === long)
    if (i < 0) continue
    const top = (node, a, b) => { const r = document.createRange(); r.setStart(node, a); r.setEnd(node, b); return Math.round(r.getBoundingClientRect().top) }
    const w = nodes[i]
    const prev = nodes[i - 1], next = nodes[i + 1]
    res.push({
      before: prev && /^[^\\p{L}\\s]+$/u.test(prev.textContent) ? { mark: prev.textContent, sameRow: Math.abs(top(prev, 0, 1) - top(w, 0, 1)) <= 3 } : null,
      after: next && /^[^\\p{L}\\s]+$/u.test(next.textContent) ? { mark: next.textContent, sameRow: Math.abs(top(next, 0, 1) - top(w, w.length - 1, w.length)) <= 3 } : null,
    })
  }
  return res
})`

const focused = (page: Page) =>
  page.evaluate(() => {
    const a = document.activeElement
    if (!a) return 'none'
    const label = a.getAttribute('aria-label') ?? a.textContent?.trim().slice(0, 30) ?? ''
    return `${a.tagName} "${label}"`
  })

// Evaluator, round 4: the patterns panel opened from the word card by a tap, then closed.
import { test } from '@playwright/test'
import * as fs from 'node:fs'
import { mockLesson } from './lesson-mocks.ts'
const EV = process.env.EVAL_OUT ?? 'C:/Users/fumna/dev/puhuscribe-v3-public/.worktrees/19-redesign/factory/jobs/19-redesign/evidence'
test('patterns panel focus', async ({ page, browserName }) => {
  const out: Record<string, unknown> = {}
  await mockLesson(page, {})
  await page.route('**/api/patterns**', (r) => r.fulfill({ contentType: 'application/json', body: JSON.stringify({ patterns: [] }) }))
  await page.goto('/session')
  await page.getByRole('button', { name: 'noi' }).first().tap()
  const card = page.getByRole('dialog').first()
  await card.waitFor()
  const link = page.getByRole('button', { name: "Patterns you've seen" })
  out['patterns control found'] = await link.count()
  if (await link.count()) {
    await link.tap()
    await page.waitForTimeout(800)
    out['open dialogs'] = await page.locator('dialog[open]').evaluateAll((d) => d.map((x) => x.getAttribute('aria-labelledby') ? document.getElementById(x.getAttribute('aria-labelledby')!)?.textContent : x.textContent?.slice(0, 30)))
    await page.keyboard.press('Escape')
    await page.waitForTimeout(500)
    out['after Escape, focus is on'] = await page.evaluate(() => `${document.activeElement?.tagName} "${document.activeElement?.textContent?.trim().slice(0, 30)}"`)
  }
  fs.writeFileSync(`${EV}/eval-r4-47-patterns-${browserName}.json`, JSON.stringify(out, null, 2))
})

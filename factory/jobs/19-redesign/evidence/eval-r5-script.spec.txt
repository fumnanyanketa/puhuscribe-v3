      const a = rects[i], b = rects[j]

      const ix = Math.min(a.right, b.right) - Math.max(a.left, b.left)
      const iy = Math.min(a.bottom, b.bottom) - Math.max(a.top, b.top)

    for (const n of nodes) {
      const own = n.parentElement.closest('button')
      const count = (() => { const r = document.createRange(); r.selectNodeContents(n); return [...r.getClientRects()].filter((q) => q.width > 1).length })()
      for (let k = 0; k < count; k++) {
        const get = () => { const r = document.createRange(); r.selectNodeContents(n); return [...r.getClientRects()].filter((q) => q.width > 1)[k] }
        let q = get()

        q = get()
        const x = (q.left + q.right) / 2, y = (q.top + q.bottom) / 2
        const hit = document.elementFromPoint(x, y)
        const hitButton = hit ? hit.closest('button') : null
        if (hitButton !== own) wrongHits.push({ text: n.textContent.slice(0, 14), piece: k, y: Math.round(y), landsOn: hitButton ? hitButton.textContent.slice(0, 14) : (hit ? hit.tagName + '.' + hit.className.toString().slice(0, 20) : 'nothing') })
      }
    }
    window.scrollTo({ top: 0, behavior: 'instant' })
    const buttons = [...line.querySelectorAll('button')].map((b) => {
      const br = b.getBoundingClientRect()
      const pieces = []
      const tw = document.createTreeWalker(b, NodeFilter.SHOW_TEXT)
      for (let n = tw.nextNode(); n; n = tw.nextNode()) {
        if (!n.textContent.trim()) continue

      }
      const rowTops = [...new Set(pieces.map((p) => Math.round(p.top)))].sort((a, b) => a - b)
      const rows = rowTops.length
      // For every row of the button after the first: how far its leftmost piece starts from the button's left.
      const rowStarts = rowTops.slice(1).map((t) => Math.min(...pieces.filter((p) => Math.round(p.top) === t).map((p) => p.left)) - br.left)

        textAlign: getComputedStyle(b).textAlign, rows,

  }

})()`

const focused = (page: Page) =>
  page.evaluate(() => {
    const a = document.activeElement
    if (!a) return 'none'
    const by = a.getAttribute('aria-labelledby')
    const label = a.getAttribute('aria-label') ?? (by ? document.getElementById(by)?.textContent : null) ?? a.textContent?.trim().slice(0, 30) ?? ''
    const role = a.getAttribute('role')
    return `${a.tagName}${role ? `[role=${role}]` : ''} "${label}"`
  })

async function closeBy(page: Page, how: string, dialog: ReturnType<Page['getByRole']>, button?: string) {
  if (how === 'button' && button) await dialog.getByRole('button', { name: button, exact: true }).tap()
  else if (how === 'Escape') await page.keyboard.press('Escape')
  else await page.touchscreen.tap(187, 12)
  await dialog.waitFor({ state: 'hidden' })
  await page.waitForTimeout(150)
}

    {
      const p = await page.context().newPage()
      watch(p)
      await mockLesson(p, { step: 'say', newWords: 3, me: { kind: 'guest' } })
      await p.goto('/session')
      await expect(p.getByRole('heading', { level: 1, name: 'Now you can say this.' })).toBeVisible()
      const chip = p.getByRole('button', { name: 'sana1' })
      const card = p.getByRole('dialog', { name: 'sana1' })
      for (const how of ['button', 'Escape', 'dimmed']) {
        await p.evaluate(() => (document.activeElement as HTMLElement | null)?.blur())
        await chip.tap()
        await card.waitFor()
        await closeBy(p, how, card, 'Close')
        res[`finish chip sana1: close by ${how === 'button' ? 'Close' : how}`] = await focused(p)
      }
      await p.close()
    }

    // 9. The first lesson tip (not a sheet).
    {
      const p = await page.context().newPage()
      watch(p)
      await mockLesson(p, { firstLesson: true })
      await p.goto('/session')
      const tip = p.getByRole('dialog', { name: 'How this screen works' })
      await tip.waitFor()
      await p.waitForTimeout(300)
      res['tip: focus while open'] = await focused(p)
      await tip.getByRole('button').first().tap()
      await tip.waitFor({ state: 'hidden' })
      await p.waitForTimeout(200)
      res['tip: after a tap on its button'] = await focused(p)
      await p.close()
    }
    res['console and page errors'] = errors
    out('47-focus-return', browserName, res)
  })

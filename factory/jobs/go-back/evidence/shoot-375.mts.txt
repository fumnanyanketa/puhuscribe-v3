// Screenshots for the go-back job at 375 x 812, from the worktree's own mocks (no database).
// Usage: node goback.mts <worktree> <out dir>
import { mkdirSync } from 'node:fs'
import { join } from 'node:path'
import { pathToFileURL } from 'node:url'

const [worktree, out] = process.argv.slice(2) as [string, string]
mkdirSync(out, { recursive: true })
const pw = await import(pathToFileURL(join(worktree, 'node_modules/@playwright/test/index.mjs')).href)
const { mockLesson, MOCK_LINES } = await import(pathToFileURL(join(worktree, 'e2e/lesson-mocks.ts')).href)
const { mockHome } = await import(pathToFileURL(join(worktree, 'e2e/road-mocks.ts')).href)
const { stubMicrophone } = await import(pathToFileURL(join(worktree, 'e2e/microphone.ts')).href)
const BASE = 'http://127.0.0.1:5194'

const browser = await pw.chromium.launch()
const json = (route: any, body: unknown, status = 200) =>
  route.fulfill({ status, contentType: 'application/json', body: JSON.stringify(body) })

async function context() {
  const ctx = await browser.newContext({ viewport: { width: 375, height: 812 }, hasTouch: true, reducedMotion: 'reduce' })
  const page = await ctx.newPage()
  page.on('pageerror', (error: Error) => console.log('page error:', error.message))
  return { ctx, page }
}

async function shot(page: any, name: string) {
  await page.waitForTimeout(400)
  await page.screenshot({ path: join(out, name + '.png') })
  console.log(name, 'scrollWidth', await page.evaluate('document.documentElement.scrollWidth'))
}

async function recordOnce(page: any) {
  await page.getByRole('button', { name: /Record yourself saying the line|Try again, record/ }).first().click()
  await page.getByRole('button', { name: /^Stop recording/ }).waitFor()
  await page.waitForTimeout(1200)
  await page.getByRole('button', { name: /^Stop recording/ }).click()
  await page.waitForTimeout(900)
}

const writes: string[] = []
function watchWrites(page: any) {
  page.on('request', (request: any) => {
    if (request.method() !== 'GET' && new URL(request.url()).pathname.startsWith('/api/')) {
      writes.push(`${request.method()} ${new URL(request.url()).pathname} practice=${String(request.postData()?.includes('"practice":true'))}`)
    }
  })
}

// 1. Previous line in Hear it, say it: a clear try on line 1, a gentle try on line 2, then back.
{
  const { ctx, page } = await context()
  await stubMicrophone(page, 'tone')
  await mockLesson(page, { step: null })
  let n = 0
  await page.route('**/api/speech/say', (route: any) => {
    n += 1
    return json(route, n === 1 ? { feedbackClass: 'clear' } : { feedbackClass: 'try_again' })
  })
  await page.goto(BASE + '/learn/session')
  await page.evaluate('document.fonts.ready')
  await page.waitForTimeout(800)
  const prev = page.getByRole('button', { name: 'Previous line' })
  console.log('line 1 before a try, Previous line shown:', await prev.count())
  await shot(page, '1a-line-1-before-try')
  await recordOnce(page)
  console.log('line 1 after a try, Previous line disabled:', await prev.isDisabled())
  await shot(page, '1a2-line-1-after-try-previous-disabled')
  await page.getByRole('button', { name: 'Next line' }).click()
  await page.waitForTimeout(400)
  await recordOnce(page)
  await shot(page, '1b-line-2-with-previous')
  await prev.click()
  await page.waitForTimeout(400)
  console.log('after previous, status has "Line 1 of 3":', await page.getByText('Line 1 of 3.').count())
  console.log('line 1 clear card kept:', await page.getByRole('region', { name: 'Your recording' }).count())
  await shot(page, '1c-back-on-line-1-kept')
  console.log('focus after back to line 1:', await page.evaluate('document.activeElement?.textContent'))
  await page.getByRole('button', { name: 'Next line' }).click()
  await page.waitForTimeout(400)
  console.log('line 2 gentle try kept (Play mine):', await page.getByRole('button', { name: 'Play mine' }).count())
  await shot(page, '1d-line-2-kept')
  await ctx.close()
}

// 2. The road: a done stop's sign opens its lessons; and the "stop done" card's button.
const LESSONS = [
  { dialogueId: 11, stop: 1, lesson: 1, title: 'Hello, I am Maria' },
  { dialogueId: 12, stop: 1, lesson: 2, title: 'Where are you from?' },
  { dialogueId: 13, stop: 1, lesson: 3, title: 'Nice to meet you' },
  { dialogueId: 14, stop: 1, lesson: 4, title: 'See you tomorrow' },
  { dialogueId: 21, stop: 2, lesson: 1, title: 'A coffee, please' },
]
{
  const { ctx, page } = await context()
  await mockHome(page, { today: { signedIn: true, today: 'continue' }, done: { 1: 4, 2: 1 } })
  await page.route('**/api/practice', (route: any) => json(route, { lessons: LESSONS }))
  await page.route('**/api/island', (route: any) => json(route, { lines: [], open: [] }))
  await page.goto(BASE + '/learn')
  await page.getByRole('list', { name: 'Stops on your road' }).waitFor()
  await page.waitForTimeout(1200)
  await shot(page, '2a-road-done-sign')
  const sign = page.getByRole('button', { name: /Practise its lessons/ }).first()
  console.log('done sign button:', await sign.getAttribute('aria-label'), await sign.boundingBox())
  await sign.click()
  await page.waitForTimeout(700)
  await shot(page, '2b-pick-a-lesson')
  await ctx.close()
}
{
  const { ctx, page } = await context()
  await mockHome(page, { today: { signedIn: true, today: 'start' }, done: { 1: 4 } })
  await page.route('**/api/practice', (route: any) => json(route, { lessons: LESSONS.slice(0, 4) }))
  await page.route('**/api/island', (route: any) => json(route, { lines: [], open: [] }))
  await page.goto(BASE + '/learn')
  await page.getByRole('list', { name: 'Stops on your road' }).waitFor()
  await page.waitForTimeout(1200)
  await shot(page, '2c-stop-done-card')
  await ctx.close()
}

// 3. A practice replay from start to finish: nothing but speaking checks may be sent.
{
  const { ctx, page } = await context()
  await stubMicrophone(page, 'tone')
  await mockLesson(page, { step: null })
  watchWrites(page)
  await page.route('**/api/speech/say', (route: any) => json(route, { feedbackClass: 'clear' }))
  await page.route('**/api/practice?dialogue=11', (route: any) =>
    json(route, {
      session: {
        state: 'ready',
        session: { id: 's1', step: 'say' },
        unit: { title: 'Unit', canDo: 'Can do.' },
        dialogue: { title: 'Dialogue' },
        lines: MOCK_LINES,
        keyLines: [1, 2],
      },
      keep: [
        {
          item: {
            formId: 1,
            speaker: 'Maria',
            gloss: 'Can I pay by card?',
            register: 'spoken',
            audio: '/audio/a.wav',
            words: [{ text: 'Voinks' }, { text: 'mä' }, { gap: true }, { text: 'kortilla', after: '?' }],
          },
          answer: { line: MOCK_LINES[1], marked: { spoken: [2], written: [2] }, form: { spoken: 'maksaa', written: 'maksaa' } },
        },
      ],
    }),
  )
  await page.goto(BASE + '/learn/session?practice=11')
  await page.evaluate('document.fonts.ready')
  await page.waitForTimeout(800)
  console.log('h1 at start:', await page.locator('h1').first().innerText())
  for (let i = 0; i < 3; i += 1) {
    await recordOnce(page)
    await page.getByRole('button', { name: i < 2 ? 'Next line' : 'Continue' }).click()
    await page.waitForTimeout(500)
  }
  console.log('h1 understand:', await page.locator('h1').first().innerText())
  await shot(page, '3a-practice-understand')
  // Understand it: try each answer row until the right one, then Next, for every question.
  for (let q = 0; q < 10; q += 1) {
    const h = (await page.locator('h1').first().innerText()).replace(/\s+/g, ' ')
    if (!/Understand/.test(h)) break
    const rows = page.locator('ul[aria-labelledby] button')
    const count = await rows.count()
    for (let r = 0; r < count; r += 1) {
      if ((await rows.nth(r).getAttribute('aria-disabled')) === 'true') continue
      await rows.nth(r).click()
      await page.waitForTimeout(150)
      if (!(await page.getByRole('button', { name: 'Next', exact: true }).isDisabled())) break
    }
    await page.getByRole('button', { name: 'Next', exact: true }).click()
    await page.waitForTimeout(500)
  }
  console.log('h1 after understand:', await page.locator('h1').first().innerText())
  await shot(page, '3b-practice-recap')
  for (let i = 0; i < 2; i += 1) {
    await recordOnce(page)
    await page.getByRole('button', { name: i < 1 ? 'Next line' : 'Continue' }).click()
    await page.waitForTimeout(500)
  }
  await page.getByRole('textbox').fill('maksaa')
  await page.getByRole('button', { name: 'Check' }).click()
  await page.waitForTimeout(400)
  await shot(page, '3c-practice-keep-it')
  await page.getByRole('button', { name: 'Next' }).click()
  await page.waitForTimeout(600)
  await shot(page, '3d-practice-finished')
  console.log('finish h1:', await page.locator('h1').first().innerText())
  console.log('writes during practice:', JSON.stringify(writes, null, 1))
  await ctx.close()
}

await browser.close()
